# JWT

`Microsoft.AspNetCore.Authentication.JwtBearer`包

盖章验证.

密钥只有服务端有,也只有服务端能验证合格.

```C#
using Microsoft.IdentityModel.Tokens;
using System.IdentityModel.Tokens.Jwt;
using System.Security.Claims;
using System.Text;

//身份信息
List<Claim> claims = new List<Claim>();
//这三个是微软的标准类型,兼容性高
claims.Add(new Claim(ClaimTypes.NameIdentifier, "6"));
claims.Add(new Claim(ClaimTypes.Name, "fs"));
claims.Add(new Claim(ClaimTypes.Role, "Admin"));
//自定义身份类型
claims.Add(new Claim("Passport", "E90000"));

//jwt密钥,最重要的
string key = "asdfasdfasdfasdfasdfasdfasdfasdf";
//加密不能直接操作string
byte[] secBytes = Encoding.UTF8.GetBytes(key);
//封装成为对称密钥
SymmetricSecurityKey secKey = new SymmetricSecurityKey(secBytes);
//用密钥和256算法生成凭证
SigningCredentials credentials = new SigningCredentials(secKey, SecurityAlgorithms.HmacSha256Signature);

//过期时间
DateTime expires=DateTime.Now.AddDays(1);
//用身份列表,过期时间,和凭证形成jwt设计类
JwtSecurityToken tokenDescriptor = new JwtSecurityToken(claims: claims, expires: expires, signingCredentials: credentials);
//这个才是真正生成出来的jwt 是string
string jwt = new JwtSecurityTokenHandler().WriteToken(tokenDescriptor);
Console.WriteLine(jwt);

//下面是服务端验证
Console.WriteLine("-----------------------");
//需要一个tokenHandler 验章机器
JwtSecurityTokenHandler tokenHandler = new JwtSecurityTokenHandler();
//验章参数
TokenValidationParameters validationParameters = new TokenValidationParameters();
//这个最为关键，告诉验章密钥，也就是jwt密钥
validationParameters.IssuerSigningKey = secKey;

//这两行没详解
//验证发布人 
validationParameters.ValidateIssuer = false;
//验证
validationParameters.ValidateAudience = false;

//用handler验证jwt是否合格，需要使用验证参数 
//out出来的是token类，可以直接.取数据
ClaimsPrincipal claimsPrincipal = tokenHandler.ValidateToken(jwt, validationParameters, out SecurityToken securityToken);

//手动取claim信息
foreach(Claim claim in claimsPrincipal.Claims)
{
    Console.WriteLine($"{claim.Type}={claim.Value}");
}
```

## WebAPI 实际配置


```C#
//Program.cs

//获得配置
builder.Services.Configure<JWTSettings>(builder.Configuration.GetSection("JWT"));
//配置服务端JWT验证，
//JwtBearerDefaults.AuthenticationScheme是常量，这里防止硬编码而已
builder.Services.AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
    .AddJwtBearer(opt =>
    {
        //获得配置密钥
        var jwtSettings = builder.Configuration.GetSection("JWT").Get<JWTSettings>();
        byte[] keyBytes = Encoding.UTF8.GetBytes(jwtSettings.SecKey);
        var secKey = new SymmetricSecurityKey(keyBytes);

        opt.TokenValidationParameters = new()
        {
            ValidateIssuer = false,
            ValidateAudience = false,
            ValidateLifetime = true,
            ValidateIssuerSigningKey = true,
            //最重要的一步，告诉服务端密钥
            IssuerSigningKey = secKey
        };
    });
```

```C#
//Controller

//允许匿名
[AllowAnonymous]
public ActionResult<string> Login(string userName,string password)
{
    if (userName == "yzk" && password == "123456")
    {
        List<Claim> claims=new List<Claim>();
        claims.Add(new Claim(ClaimTypes.NameIdentifier, "1"));
        claims.Add(new Claim(ClaimTypes.Name, userName));

        //这里由依赖注入得到
        string key = jwtSettingsOpt.Value.SecKey;
        DateTime expire = DateTime.Now.AddSeconds(jwtSettingsOpt.Value.ExpireSeconds);

        byte[] secBytes = Encoding.UTF8.GetBytes(key);
        var secKey = new SymmetricSecurityKey(secBytes);
        var credentials = new SigningCredentials(secKey, SecurityAlgorithms.HmacSha256Signature);
        var tokenDescriptor = new JwtSecurityToken(claims: claims, expires: expire, signingCredentials: credentials);
        string jwt=new JwtSecurityTokenHandler().WriteToken(tokenDescriptor);
        return jwt;
    }
    else
    {
        return BadRequest();
    }
}
```

要启动验证需要在控制器或者方法上标记`[Authorize]`,如果在控制器上标记则里面所有方法都需要验证，除非在不需要验证的方法上标记`[AllowAnonymous]`.

得到的JWT需要在请求头里面以`Authorization`:`Bearer YourJWT`的方式传递.

```C#
//有了JWT可以解析出其中的数据
[Authorize]
public ActionResult<string> Index()
{
    Claim? claim = User.FindFirst(ClaimTypes.Name);
    return "hello"+claim.Value;
}
```
